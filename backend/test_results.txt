WITH InventoryValueCTE AS (
    SELECT 
        DP.ItemID,
        DP.ItemName,
        SUM(FIS.QuantityOnHand) AS TotalQtyOnHand,
        SUM(FIS.InventoryValue) AS TotalInventoryValue
    FROM FactInventorySnapshot FIS
    JOIN DimProduct DP ON FIS.ProductKey = DP.ProductKey
    WHERE DP.IsDiscontinued = 0
    GROUP BY DP.ItemID, DP.ItemName
),
ABC_Categorized AS (
    SELECT 
        ItemID,
        ItemName,
        TotalQtyOnHand,
        TotalInventoryValue,
        SUM(TotalInventoryValue) OVER (ORDER BY TotalInventoryValue DESC ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS RunningTotal,
        SUM(TotalInventoryValue) OVER () AS GrandTotal
    FROM InventoryValueCTE
),
ABC_Final AS (
    SELECT 
        ItemID,
        ItemName,
        TotalQtyOnHand,
        TotalInventoryValue,
        CASE 
            WHEN RunningTotal / NULLIF(GrandTotal,0) <= 0.8 THEN 'A'
            WHEN RunningTotal / NULLIF(GrandTotal,0) <= 0.95 THEN 'B'
            ELSE 'C'
        END AS ABC_Category
    FROM ABC_Categorized
)
SELECT TOP 40
    ABC_Category,
    SUM(TotalInventoryValue) AS TotalInventoryValue,
    SUM(TotalQtyOnHand) AS TotalQtyOnHand
FROM ABC_Final
GROUP BY ABC_Category
ORDER BY CASE ABC_Category WHEN 'A' THEN 1 WHEN 'B' THEN 2 ELSE 3 END

        DP.ItemID,

    FROM FactInventorySnapshot FIS

),
ABC_Categorized AS (
    SELECT 
        ItemID,
        ItemName,
        TotalQtyOnHand,
        TotalInventoryValue,
        SUM(TotalInventoryValue) OVER (ORDER BY TotalInventoryValue DESC ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS RunningTotal,
        SUM(TotalInventoryValue) OVER () AS GrandTotal,
        CASE 
            WHEN SUM(TotalInventoryValue) OVER (ORDER BY TotalInventoryValue DESC ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) <= 0.8 * SUM(TotalInventoryValue) OVER () THEN 'A'
            WHEN SUM(TotalInventoryValue) OVER (ORDER BY TotalInventoryValue DESC ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) <= 0.95 * SUM(TotalInventoryValue) OVER () THEN 'B'

    FROM InventoryValueCTE
)
, RankedItems AS (
    SELECT *,
        ROW_NUMBER() OVER (PARTITION BY ABC_Category ORDER BY TotalInventoryValue DESC) AS rn
    FROM ABC_Categorized
)
SELECT TOP 15
    ABC_Category,
    ItemID,
    ItemName,
    TotalQtyOnHand,
    TotalInventoryValue
FROM RankedItems

[P1-07] How ABC categories are assigned to items for ABC Analysis, what is rat
Q: How ABC categories are assigned to items for ABC Analysis, what is rationale
================================================================================
SQL:
(none)

RESPONSE:
ABC categories in inventory analysis are assigned based on the relative value contribution of each item to the total inventory value:

1. **A Category**: These are the top items that account for roughly 80% of the total inventory value, even though they may represent only about 10-20% of the total items. They are high-value, tightly controlled, and require frequent review.

2. **B Category**: These items typically account for the next 15% of inventory value and represent about 20-30% of the items. They are of moderate value and importance, with less strict controls than A items.

3. **C Category**: These are the remaining items, often making up about 5% of the inventory value but 50% or more of the item count. They are low-value, require the simplest controls, and are managed in bulk.

**Rationale:**  
The ABC classification helps prioritize management focus and resources:
- **A items** need tight inventory control, accurate records, and frequent reordering.
- **B items** require moderate attention.
- **C items** can be managed with simpler, less costly methods.

This approach ensures that the most valuable inventory receives the most attention, optimizing working capital and reducing carrying costs.

[P1-10] Provide monthly sales for 2026 vs 2025 for TJ Maxx, Marshalls, Mackenz
Q: Provide monthly sales for 2026 vs 2025 for TJ Maxx, Marshalls, Mackenzie Childs, and Williams Sonoma. Calculate the gap to reach 2025 monthly averages and project annual recovery value.
================================================================================
SQL:
WITH TargetCustomers AS (
    SELECT CustomerKey, CustomerName
    FROM DimCustomer
    WHERE CustomerName IN ('TJ Maxx', 'Marshalls', 'Mackenzie Childs', 'Williams Sonoma')
),
Sales2025 AS (
    SELECT 
        FSI.CustomerKey,
        DD.Month,
        DD.MonthName,
        SUM(FSI.MerchandiseAmount) AS Revenue2025
    FROM FactSalesInvoice FSI
    JOIN DimDate DD ON FSI.DateKey = DD.DateKey
    WHERE DD.Year = 2025
      AND FSI.Status NOT IN ('Void', 'Cancelled', 'Reversed')
      AND FSI.CustomerKey IN (SELECT CustomerKey FROM TargetCustomers)
    GROUP BY FSI.CustomerKey, DD.Month, DD.MonthName
),
Sales2026 AS (
    SELECT 
        FSI.CustomerKey,
        DD.Month,
        DD.MonthName,
        SUM(FSI.MerchandiseAmount) AS Revenue2026
    FROM FactSalesInvoice FSI
    JOIN DimDate DD ON FSI.DateKey = DD.DateKey
    WHERE DD.Year = 2026
      AND FSI.Status NOT IN ('Void', 'Cancelled', 'Reversed')
      AND FSI.CustomerKey IN (SELECT CustomerKey FROM TargetCustomers)
    GROUP BY FSI.CustomerKey, DD.Month, DD.MonthName
),
MonthlyAvg2025 AS (
    SELECT CustomerKey, AVG(Revenue2025) AS MonthlyAvg2025
    FROM Sales2025
    GROUP BY CustomerKey
),
Combined AS (
    SELECT 
        TC.CustomerName,
        S26.Month,
        S26.MonthName,
        ISNULL(S26.Revenue2026, 0) AS Revenue2026,
        ISNULL(S25.Revenue2025, 0) AS Revenue2025,
        ISNULL(MA.MonthlyAvg2025, 0) AS MonthlyAvg2025,
        (ISNULL(S26.Revenue2026, 0) - ISNULL(MA.MonthlyAvg2025, 0)) AS GapTo2025Avg
    FROM TargetCustomers TC
    CROSS JOIN (SELECT DISTINCT Month, MonthName FROM DimDate WHERE Year = 2026) M
    LEFT JOIN Sales2026 S26 ON TC.CustomerKey = S26.CustomerKey AND S26.Month = M.Month
    LEFT JOIN Sales2025 S25 ON TC.CustomerKey = S25.CustomerKey AND S25.Month = M.Month
    LEFT JOIN MonthlyAvg2025 MA ON TC.CustomerKey = MA.CustomerKey
)
SELECT TOP 40
    CustomerName,
    Month,
    MonthName,
    Revenue2026,
    MonthlyAvg2025,
    (MonthlyAvg2025 - Revenue2026) AS RecoveryNeeded
FROM Combined
ORDER BY CustomerName, Month

[P1-48] Prepare an ABC Analysis summary of items on hand , value and quantity,
Q: Prepare an ABC Analysis summary of items on hand , value and quantity, RETURN OUTPUT IN Categories, Quantities and Inventory values

WITH Inventory AS (
    SELECT 
        DP.ProductKey,
        DP.ItemID,
        DP.ItemName,
        SUM(FIS.QuantityOnHand) AS TotalQtyOnHand,
        SUM(FIS.InventoryValue) AS TotalInventoryValue
    FROM FactInventorySnapshot FIS
    JOIN DimProduct DP ON FIS.ProductKey = DP.ProductKey
    WHERE DP.IsDiscontinued = 0
    GROUP BY DP.ProductKey, DP.ItemID, DP.ItemName
),
RankedInventory AS (
    SELECT *,
        SUM(TotalInventoryValue) OVER (ORDER BY TotalInventoryValue DESC ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS RunningValue,
        SUM(TotalInventoryValue) OVER () AS TotalValue
    FROM Inventory
),
ABC_Categorized AS (
    SELECT *,
        CASE 
            WHEN RunningValue <= 0.8 * TotalValue THEN 'A'
            WHEN RunningValue <= 0.95 * TotalValue THEN 'B'
            ELSE 'C'
        END AS ABC_Category
    FROM RankedInventory
)
SELECT TOP 40
    ABC_Category,
    SUM(TotalQtyOnHand) AS TotalQuantity,
    SUM(TotalInventoryValue) AS TotalInventoryValue
FROM ABC_Categorized
GROUP BY ABC_Category
ORDER BY CASE ABC_Category WHEN 'A' THEN 1 WHEN 'B' THEN 2 ELSE 3 END

        FSI.CustomerKey,

    JOIN DimDate DD ON FCM.InvoiceDateKey = DD.DateKey

[P1-50] Analyze customer payment habits for the 2024 and 2025. Categories base
Q: Analyze customer payment habits for the 2024 and 2025. Categories based on the how much time they usually take to make payments, What is best time to proceed collections
================================================================================
SQL:
WITH PaymentLag AS (
    SELECT 
        FCP.CustomerKey,
        DATEDIFF(DAY, MIN(DD_Doc.FullDate), MIN(DD_Payment.FullDate)) AS DaysToPay,
        COUNT(DISTINCT FCP.CashReceiptNo) AS PaymentCount
    FROM FactCustomerPayment FCP
    JOIN DimDate DD_Doc ON FCP.DocDateKey = DD_Doc.DateKey
    JOIN DimDate DD_Payment ON FCP.PaymentDateKey = DD_Payment.DateKey
    WHERE DD_Doc.Year IN (2024, 2025)
      AND FCP.Status NOT IN ('Bounced', 'Void')
      AND FCP.CustomerKey IS NOT NULL
    GROUP BY FCP.CustomerKey, DD_Doc.Year, DD_Doc.Month
),
CustomerAvgLag AS (
    SELECT 
        PL.CustomerKey,
        AVG(CAST(PL.DaysToPay AS FLOAT)) AS AvgDaysToPay,
        SUM(PL.PaymentCount) AS TotalPayments
    FROM PaymentLag PL
    GROUP BY PL.CustomerKey
),
Bucketed AS (
    SELECT 
        CAL.CustomerKey,
        CASE 
            WHEN CAL.AvgDaysToPay <= 0 THEN 'Early'
            WHEN CAL.AvgDaysToPay <= 5 THEN 'On-Time'
            ELSE 'Late'
        END AS PaymentCategory,
        CAL.AvgDaysToPay,
        CAL.TotalPayments
    FROM CustomerAvgLag CAL
)
SELECT TOP 40
    B.PaymentCategory,
    COUNT(DC.CustomerID) AS CustomerCount,
    AVG(B.AvgDaysToPay) AS AvgDaysToPay,
    SUM(B.TotalPayments) AS TotalPayments
FROM Bucketed B
JOIN DimCustomer DC ON B.CustomerKey = DC.CustomerKey
GROUP BY B.PaymentCategory
ORDER BY CASE B.PaymentCategory WHEN 'Early' THEN 1 WHEN 'On-Time' THEN 2 ELSE 3 END
